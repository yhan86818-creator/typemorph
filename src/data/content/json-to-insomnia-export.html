JSON to Insomnia Export: Workspace Format, Environment Chaining, and Request Templates

Insomnia's export format (v4) organizes everything — workspaces, request groups, individual requests, environments, and environment templates — into a flat resources array linked by _id references. Understanding this structure lets you generate complete workspaces programmatically: from an OpenAPI spec, from a list of API endpoints, or from JSON request/response samples. The distinctive features of Insomnia exports are chained responses (response from request A becomes input to request B), template tags ({% response %}, {% env %}), and sub-environments that overlay the base environment for staging vs production configuration.

Response Chaining with Template Tags

{
  "_id":    "req_get_product",
  "_type":  "request",
  "name":   "Get Product (using created ID)",
  "method": "GET",
  "url":    "{{ _.baseUrl }}/products/{% response 'body', 'req_create_product', '$.id', 'when-expired', 60 %}",
  // ↑ Extracts the 'id' field from the body of req_create_product
  // Insomnia template tag syntax: {% response 'attribute', 'requestId', 'jsonpath', 'trigger', 'maxAge' %}
  "headers": [
    { "name": "X-API-Key", "value": "{{ _.apiKey }}", "enabled": true }
  ]
}

// Other useful template tags:
// {% env 'API_KEY' %}               — value from environment variable
// {% timestamp %}                   — current Unix timestamp
// {% now 'iso-8601' %}              — current datetime
// {% uuid %}                        — random UUID
// {% randomInt 1, 1000 %}           — random integer
// {% base64 'encode', 'username:password' %}  — base64 encode

// OAuth2 token extraction (login → use token in all subsequent requests):
{
  "_id":    "req_login",
  "_type":  "request",
  "name":   "Login",
  "method": "POST",
  "url":    "{{ _.baseUrl }}/auth/login",
  "body": {
    "mimeType": "application/json",
    "text":     "{\"email\": \"{{ _.email }}\", \"password\": \"{{ _.password }}\"}"
  }
}

// In a subsequent request, reference the token from the login response:
{
  "_id":    "req_get_profile",
  "_type":  "request",
  "name":   "Get Profile",
  "method": "GET",
  "url":    "{{ _.baseUrl }}/users/me",
  "authentication": {
    "type":   "bearer",
    "token":  "{% response 'body', 'req_login', '$.access_token', 'when-expired', 3600 %}"
  }
}

Programmatic Collection Generation

// Generate an Insomnia v4 export from an OpenAPI spec
import yaml   from 'js-yaml';
import { v4 as uuidv4 } from 'uuid';

interface InsomniaResource {
  _id:    string;
  _type:  string;
  [key: string]: unknown;
}

function openApiToInsomnia(spec: any): object {
  const workspaceId = `wrk_${uuidv4().replace(/-/g, '').slice(0, 12)}`;
  const resources: InsomniaResource[] = [
    {
      _id:         workspaceId,
      _type:       'workspace',
      name:        spec.info.title,
      description: spec.info.description ?? '',
      scope:       'collection',
    },
    {
      _id:      `env_base_${workspaceId}`,
      _type:    'environment',
      parentId: workspaceId,
      name:     'Base Environment',
      data: {
        baseUrl: spec.servers?.[0]?.url ?? 'http://localhost:3000',
      },
    },
  ];

  // Group by tags
  const tagGroups: Record<string, string> = {};

  for (const [path, methods] of Object.entries(spec.paths ?? {})) {
    for (const [method, operation] of Object.entries(methods as any)) {
      if (!['get','post','put','patch','delete'].includes(method)) continue;

      const op        = operation as any;
      const tag       = op.tags?.[0] ?? 'Default';
      const groupKey  = `grp_${tag.replace(/\W/g, '_').toLowerCase()}`;

      if (!tagGroups[tag]) {
        tagGroups[tag] = groupKey;
        resources.push({ _id: groupKey, _type: 'request_group', parentId: workspaceId, name: tag });
      }

      const reqId = `req_${op.operationId ?? uuidv4().slice(0, 8)}`;
      const headers: any[] = [{ name: 'Accept', value: 'application/json', enabled: true }];
      if (['post','put','patch'].includes(method)) {
        headers.push({ name: 'Content-Type', value: 'application/json', enabled: true });
      }

      const exampleBody = op.requestBody?.content?.['application/json']?.example;

      resources.push({
        _id:      reqId,
        _type:    'request',
        parentId: tagGroups[tag],
        name:     op.summary ?? `${method.toUpperCase()} ${path}`,
        method:   method.toUpperCase(),
        url:      `{{ _.baseUrl }}${path}`,
        headers,
        body:     exampleBody ? { mimeType: 'application/json', text: JSON.stringify(exampleBody, null, 2) } : undefined,
      });
    }
  }

  return {
    _type:             'export',
    __export_format:   4,
    __export_date:     new Date().toISOString(),
    __export_source:   'generated',
    resources,
  };
}

import { writeFileSync } from 'fs';
const spec    = yaml.load(readFileSync('./openapi.yaml', 'utf-8'));
const exported = openApiToInsomnia(spec);
writeFileSync('./insomnia-collection.json', JSON.stringify(exported, null, 2));

Best Practices for Production

  • Store credentials in sub-environments, not the base environment: The base environment should have structure only (baseUrl, apiVersion). Sub-environments (Staging, Production) hold actual credentials. This lets you commit the base environment to Git while keeping the sub-environments (with real API keys) in a secrets manager.
  • Use response chaining for multi-step workflows: Insomnia's {% response %} tag is the key feature that distinguishes it from basic HTTP clients. A login → create → verify workflow where each step uses the previous response's data tests your entire authentication flow in one collection run.
  • Use Insomnia's Git Sync for team collaboration: Insomnia has built-in Git repository sync. Store your workspace export in a insomnia/ directory in your API repository. Engineers clone the repo, import the workspace, and immediately have a correctly configured API client — no manual setup.
  • Generate the export from your OpenAPI spec, not from the UI: Building the Insomnia workspace by clicking through the UI doesn't scale for APIs with 50+ endpoints. Generate it from the OpenAPI spec using a script and re-generate when the spec changes.

FAQ

Q: What's the difference between Insomnia and Postman for modern teams?
A: Insomnia is lighter with a cleaner UI and better GraphQL support. Postman has more enterprise features: team collaboration, API monitoring, and a larger ecosystem of community collections. Insomnia's export format is simpler to generate programmatically. For solo developers or small teams, Insomnia's simplicity is an advantage. For large teams with test automation requirements, Postman's Newman CLI integration is more mature.

Q: Can I run Insomnia tests in CI like Newman does for Postman?
A: Yes, using the Inso CLI (npm install inso). Run inso run tests to execute test suites from an Insomnia collection in CI. It supports the same export format and produces JUnit-compatible XML output for CI test reporting.

Q: How do I handle self-signed SSL certificates in Insomnia?
A: In Insomnia Settings → Preferences → Request/Response, disable SSL validation globally. For per-request control, add a client certificate. For proxied requests through mitmproxy or Charles, configure the proxy under Application → Preferences and add the proxy's CA certificate.
