JSON to Tailwind Config: Design Tokens, TypeScript Config, and Figma Integration

Tailwind CSS is configured entirely in JavaScript (or TypeScript), which means JSON design tokens feed directly into the config without a build step. The pattern matters: importing raw token JSON and spreading it into theme.extend works for simple projects, but at scale you need a structured mapping — transforming W3C Design Token format, generating color shade scales from base colors, wiring up CSS custom properties for runtime theming, and keeping TypeScript strict enough that accessing a non-existent token is a compile error. This guide covers the complete pipeline from design tool JSON export to a production-ready Tailwind config.

Live Example: Design Tokens → Tailwind Config

// design-tokens.json (W3C Design Tokens format)
{

    "brand": {
      "primary":   { "$value": "#2563eb", "$type": "color" },
      "secondary": { "$value": "#7c3aed", "$type": "color" },
      "accent":    { "$value": "#db2777", "$type": "color" }
    },
    "neutral": {
      "50":  { "$value": "#f9fafb", "$type": "color" },

      "900": { "$value": "#111827", "$type": "color" }
    },
    "semantic": {
      "success": { "$value": "#16a34a", "$type": "color" },
      "error":   { "$value": "#dc2626", "$type": "color" },
      "warning": { "$value": "#d97706", "$type": "color" }
    }
  },
  "spacing": {
    "xs":  { "$value": "0.25rem", "$type": "dimension" },
    "sm":  { "$value": "0.5rem",  "$type": "dimension" },
    "md":  { "$value": "1rem",    "$type": "dimension" },
    "lg":  { "$value": "1.5rem",  "$type": "dimension" },
    "xl":  { "$value": "2rem",    "$type": "dimension" },
    "2xl": { "$value": "3rem",    "$type": "dimension" }
  },
  "typography": {
    "fontFamily": {
      "sans":  { "$value": "\"Inter\", system-ui, sans-serif",  "$type": "fontFamily" },
      "mono":  { "$value": "\"JetBrains Mono\", monospace",     "$type": "fontFamily" }
    },
    "fontSize": {
      "xs":   { "$value": "0.75rem",  "$type": "dimension" },
      "sm":   { "$value": "0.875rem", "$type": "dimension" },
      "base": { "$value": "1rem",     "$type": "dimension" },
      "xl":   { "$value": "1.25rem",  "$type": "dimension" },
      "2xl":  { "$value": "1.5rem",   "$type": "dimension" },
      "4xl":  { "$value": "2.25rem",  "$type": "dimension" }
    }
  }
}

// tailwind.config.ts
import type { Config } from 'tailwindcss';
import tokens from './design-tokens.json';

// Helper: extract $value from W3C Design Token format
function extractValues(obj: Record<string, any>): Record<string, string> {
  return Object.fromEntries(
    Object.entries(obj).map(([key, val]) => [
      key,
      typeof val === 'object' && '$value' in val ? val.$value : val
    ])
  );
}

const config: Config = {
  content: ['./src/**/*.{ts,tsx,html}'],
  theme: {
    extend: {
      colors: {
        brand:   extractValues(tokens.color.brand),
        neutral: extractValues(tokens.color.neutral),
        success: tokens.color.semantic.success.$value,
        error:   tokens.color.semantic.error.$value,
        warning: tokens.color.semantic.warning.$value,
      },
      spacing:    extractValues(tokens.spacing),
      fontFamily: extractValues(tokens.typography.fontFamily),
      fontSize:   extractValues(tokens.typography.fontSize),
    },
  },
};

export default config;

Generating Color Shade Scales

// Most design systems need 50–950 shade scales, not just one base color.
// Use a library to generate shades from a base color.

// Option 1: tailwindcss/colors as reference scale
import colors from 'tailwindcss/colors';

// Option 2: generate shades with chroma-js
import chroma from 'chroma-js';

function generateShades(baseColor: string): Record<string, string> {
  const scale = chroma.scale(['white', baseColor, 'black'])
    .mode('lch')   // perceptually uniform color space
    .colors(11);   // 11 steps

  const stops = [50, 100, 200, 300, 400, 500, 600, 700, 800, 900, 950];
  return Object.fromEntries(stops.map((stop, i) => [stop, scale[i]]));
}

// tailwind.config.ts
const config: Config = {
  theme: {
    extend: {
      colors: {
        brand: {
          // Auto-generates: brand-50, brand-100 ... brand-950
          ...generateShades(tokens.color.brand.primary.$value),
        },
        purple: generateShades(tokens.color.brand.secondary.$value),
      }
    }
  }
};

// Usage in markup:
// bg-brand-500 text-brand-50 border-brand-700
// dark:bg-brand-900 dark:text-brand-100

CSS Custom Properties for Runtime Theming

// CSS variables enable runtime theme switching (dark mode, white-labeling)
// without rebuilding CSS. Tailwind 3 supports this via arbitrary values
// and v4 makes it the default approach.

// globals.css
:root {
  --color-brand:     #2563eb;
  --color-bg:        #ffffff;
  --color-text:      #111827;
  --color-surface:   #f3f4f6;
}

[data-theme="dark"] {
  --color-brand:     #3b82f6;
  --color-bg:        #111827;
  --color-text:      #f9fafb;
  --color-surface:   #1f2937;
}

// tailwind.config.ts — reference CSS variables
const config: Config = {
  theme: {
    extend: {
      colors: {
        brand:   'rgb(var(--color-brand) / <alpha-value>)',  // opacity support
        bg:      'var(--color-bg)',
        text:    'var(--color-text)',
        surface: 'var(--color-surface)',
      }
    }
  }
};

// Usage: colors work with opacity modifiers
// bg-brand/50 → background-color: rgb(var(--color-brand) / 0.5)
// text-text → color: var(--color-text) (switches with theme)

TypeScript Theme Typing with satisfies

// Use TypeScript 4.9+ satisfies operator to type-check the config
// while preserving the literal type (not widening to string)

import type { ThemeConfig } from 'tailwindcss/types/config';

const colors = {
  brand:   '#2563eb',
  accent:  '#7c3aed',
  success: '#16a34a',
  error:   '#dc2626',
} satisfies ThemeConfig['colors'];

const config: Config = {
  theme: {
    extend: {
      colors,   // colors is typed — hover reveals exact values
    }
  }
};

// Custom plugin with typed tokens
import plugin from 'tailwindcss/plugin';

const tokenPlugin = plugin(({ addBase, addComponents, theme }) => {
  addBase({
    ':root': {
      '--color-brand':   theme('colors.brand') as string,
      '--color-accent':  theme('colors.accent') as string,
      '--spacing-md':    theme('spacing.md') as string,
    }
  });

  addComponents({
    '.btn-primary': {
      backgroundColor: theme('colors.brand'),
      color:          '#ffffff',
      padding:        `${theme('spacing.sm')} ${theme('spacing.md')}`,
      borderRadius:   theme('borderRadius.md') as string,
      fontFamily:     theme('fontFamily.sans') as string,
      '&:hover': {
        backgroundColor: theme('colors.brand-dark', '#1d4ed8'),
      }
    }
  });
});

Figma Tokens → JSON → Tailwind Pipeline

// 1. Export from Figma using Tokens Studio for Figma plugin
//    → produces W3C Design Token format JSON

// 2. Transform tokens with Style Dictionary
//    npm install -D style-dictionary

// config.sd.js
const StyleDictionary = require('style-dictionary');

StyleDictionary.buildAllPlatforms({
  source: ['tokens/**/*.json'],
  platforms: {
    tailwind: {
      buildPath: 'src/config/',
      files: [{
        destination: 'tokens.ts',
        format: 'javascript/es6',
        filter: { attributes: { category: 'color' } },
      }]
    },
    css: {
      buildPath: 'src/styles/',
      files: [{ destination: 'variables.css', format: 'css/variables' }]
    }
  }
});

// 3. Import generated tokens into tailwind.config.ts
import { colorTokens, spacingTokens } from './src/config/tokens';

// 4. CI step to detect token drift
// In package.json:
{
  "scripts": {
    "tokens:export": "figma-export --token $FIGMA_TOKEN --file $FIGMA_FILE > tokens/figma.json",
    "tokens:build":  "style-dictionary build --config config.sd.js",
    "tokens:sync":   "npm run tokens:export && npm run tokens:build && git diff --exit-code"
  }
}

Best Practices for Production

  • Use theme.extend to add tokens, not theme to replace: Setting theme.colors directly removes all of Tailwind's built-in colors (gray, red, blue, etc.). Use theme.extend.colors to add your tokens alongside the defaults. Only use theme.colors if you deliberately want to eliminate all default utilities and ship only your brand tokens.
  • Set content paths precisely: Tailwind's JIT engine scans your source files for class names. Over-broad patterns ('**/*') slow down compilation; too narrow patterns miss dynamically constructed class names. Include every directory with template files and exclude node_modules and build output.
  • Use CSS variables for dark mode tokens: The class dark mode strategy (darkMode: 'class') combined with CSS custom properties gives you runtime theme switching without a new build. Set token values in :root and override in .dark — one set of Tailwind utilities serves both themes.
  • Lint your design token keys: Inconsistent naming (primary vs brand-primary vs brand.primary) breaks autocomplete and causes confusion. Pick a convention (flat keys like brand-500, or nested like { brand: { 500: '#...' } }) and enforce it in the token source with a JSON Schema or custom lint rule.

FAQ

Q: What is the difference between Tailwind v3 and v4 config?
A: Tailwind v4 moves to a CSS-first configuration (@import "tailwindcss"; @theme { --color-brand: #2563eb; } directly in CSS) instead of a JavaScript config file. Design tokens defined as CSS custom properties are automatically available as utility classes. The JavaScript config is still supported for complex transformations, but the primary config moves to CSS.

Q: How do I import JSON design tokens into a TypeScript Tailwind config?
A: Add "resolveJsonModule": true to tsconfig.json, then import tokens from './tokens.json'. The JSON is type-checked at compile time. For the config file itself, name it tailwind.config.ts and run Tailwind via a TS loader or with ts-node.

Q: How do I handle opacity with CSS variable colors?
A: Use the rgb(var(--color-brand) / <alpha-value>) pattern in the config. This tells Tailwind to generate rgb() syntax with the <alpha-value> placeholder, which Tailwind replaces with the opacity value. Without this, bg-brand/50 won't work with CSS variable colors.
