JSON to Docker Compose: Healthchecks, Networks, and Multi-Service Orchestration

Docker Compose defines multi-container applications as a single YAML document — but the YAML is just a serialization of a well-defined schema that maps directly from JSON. Understanding the full Compose spec lets you generate correct, production-grade configurations: healthchecks that gate depends_on conditions, named volumes with explicit drivers, custom networks for service isolation, resource limits for production deployments, and profiles to optionally include monitoring tools. The difference between a generated Compose file that works and one that causes subtle issues in CI is usually in these details.

Live Example: Full Stack Application

// Input JSON (application service spec)
{
  "project": "myapp",
  "services": [
    { "name": "web",    "image": "myapp/web:latest",    "port": 3000, "depends_on": ["api"] },
    { "name": "api",    "image": "myapp/api:latest",    "port": 8080, "depends_on": ["db", "redis"] },
    { "name": "worker", "image": "myapp/worker:latest",              "depends_on": ["db", "redis"] },
    { "name": "db",     "image": "postgres:16-alpine",   "port": 5432 },
    { "name": "redis",  "image": "redis:7-alpine",       "port": 6379 }
  ]
}

# Generated docker-compose.yml
name: myapp

services:
  web:
    image: myapp/web:latest
    ports:
      - "3000:3000"
    environment:
      API_URL: http://api:8080
    depends_on:
      api:
        condition: service_healthy   # wait until API passes healthcheck
    restart: unless-stopped
    networks:
      - frontend

  api:
    image: myapp/api:latest
    ports:
      - "8080:8080"
    environment:
      DATABASE_URL: postgresql://postgres:${DB_PASSWORD}@db:5432/myapp
      REDIS_URL: redis://redis:6379
    env_file:
      - .env                         # load secrets from file, not inline
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s              # grace period before health checks begin
    restart: unless-stopped
    networks:
      - frontend
      - backend

  worker:
    image: myapp/worker:latest
    environment:
      DATABASE_URL: postgresql://postgres:${DB_PASSWORD}@db:5432/myapp
      REDIS_URL: redis://redis:6379
    env_file:
      - .env
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - backend
    deploy:
      replicas: 2                    # run 2 worker instances
      resources:
        limits:
          cpus: '0.5'
          memory: 512M

  db:
    image: postgres:16-alpine
    ports:
      - "5432:5432"                  # expose for local dev tools only
    environment:
      POSTGRES_DB:       myapp
      POSTGRES_USER:     postgres
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - db_data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql:ro  # init script
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d myapp"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - backend

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD}", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - backend

volumes:
  db_data:     { driver: local }
  redis_data:  { driver: local }

networks:
  frontend:    { driver: bridge }    # web ↔ api
  backend:     { driver: bridge }    # api, worker ↔ db, redis

Healthchecks and depends_on Conditions

# Without conditions, depends_on only waits for the container to START — not be READY
# This causes race conditions where the app starts before the DB accepts connections

# WRONG: no condition (starts after container starts, not after DB is ready)
depends_on:
  - db

# CORRECT: waits until healthcheck passes
depends_on:
  db:
    condition: service_healthy

# Three available conditions:
# service_started  — default: container has started (not useful for wait logic)
# service_healthy  — healthcheck reports healthy (requires healthcheck in the target)
# service_completed_successfully — the service exited with code 0 (for init containers)

# Database healthcheck patterns:
# PostgreSQL:
healthcheck:
  test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
  interval: 10s
  timeout: 5s
  retries: 5

# MySQL:
healthcheck:
  test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "--silent"]

# MongoDB:
healthcheck:
  test: ["CMD", "mongosh", "--eval", "db.adminCommand('ping')"]

Build Context vs Image

# For local development: build from Dockerfile
services:
  api:
    build:
      context: ./api              # path to Dockerfile
      dockerfile: Dockerfile.dev  # optional override
      args:
        NODE_VERSION: "20"
        BUILD_ENV: development
    image: myapp/api:dev          # tag the built image (optional)

  # Multi-stage build — only build the stage you need
  api-prod:
    build:
      context: ./api
      target: production          # stops at the 'production' stage in Dockerfile

# Override for CI: build from CI-built image, not local source
# docker compose -f docker-compose.yml -f docker-compose.ci.yml up
# docker-compose.ci.yml:
services:
  api:
    image: myapp/api:${GIT_SHA}  # from CI registry
    build: !reset {}              # remove build config

Profiles: Optional Services

# Profiles let you conditionally include services
# Not included by default — only start when explicitly requested

services:
  # Core services (no profile — always active)
  api: { image: myapp/api:latest }
  db:  { image: postgres:16-alpine }

  # Monitoring (only when --profile monitoring)
  prometheus:
    image: prom/prometheus:latest
    profiles: [monitoring]
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml

  grafana:
    image: grafana/grafana:latest
    profiles: [monitoring]
    ports: ["3001:3000"]

  # Debug tools (only when --profile debug)
  pgadmin:
    image: dpage/pgadmin4
    profiles: [debug]
    ports: ["5050:80"]

# Usage:
docker compose up                          # core only
docker compose --profile monitoring up     # core + monitoring
docker compose --profile monitoring --profile debug up  # everything

Secrets Management

# Method 1: .env file (local dev only — NEVER commit with real values)
# .env
DB_PASSWORD=dev_password_change_me
REDIS_PASSWORD=dev_redis_pw

# Method 2: Docker Secrets (Swarm/production)
services:
  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_PASSWORD_FILE: /run/secrets/db_password
    secrets:
      - db_password

secrets:
  db_password:
    file: ./secrets/db_password.txt   # or: external: true (from Swarm secret)

# Method 3: Read from environment (CI/CD injects into shell)
services:
  api:
    environment:
      DATABASE_URL: ${DATABASE_URL}   # CI sets DATABASE_URL in env before running compose

Best Practices for Production

  • Always add healthchecks to stateful services: PostgreSQL, MySQL, Redis, and MongoDB all need healthchecks so that depends_on: condition: service_healthy works correctly. Without healthchecks, services that depend on the database start before it's ready and crash with connection errors.
  • Use named volumes, not bind mounts, for persistent data: Bind mounts (./data:/var/lib/postgresql/data) depend on the host directory existing and have permission issues across environments. Named volumes (db_data:/var/lib/postgresql/data) are managed by Docker, portable across machines, and can be backed up with docker volume ls.
  • Use separate networks for service isolation: Putting all services on a single default network means every service can reach every other service by name. Create separate frontend and backend networks so that, e.g., the database is only reachable from the API and worker, not from the web frontend container.
  • Use restart: unless-stopped not always: always restarts even if you explicitly stop the container with docker stop. unless-stopped respects explicit stops and only auto-restarts on failure or system reboot.

FAQ

Q: What is the difference between docker-compose (v1) and docker compose (v2)?
A: Docker Compose v2 (docker compose) is a Go rewrite built into the Docker CLI as a plugin — it's faster, uses the same config format, and is the current standard. Docker Compose v1 (docker-compose, Python) is deprecated. If you see version: '3.8' at the top of a Compose file, that's valid but the version key itself is deprecated in v2 spec — you can omit it.

Q: Can I use Docker Compose in production?
A: Yes, for single-host deployments (one server, multiple containers). For multi-host orchestration, use Docker Swarm (docker stack deploy) or Kubernetes. Docker Compose with Docker Swarm mode accepts the same docker-compose.yml syntax with the deploy key for replicas, resource limits, and update policies.

Q: How do I run one-time database migrations?
A: Use a service with command override and restart: "no", or profiles: [migrate]. Run it with docker compose run --rm api npm run migrate before starting the main services. The --rm flag removes the container after it exits.
